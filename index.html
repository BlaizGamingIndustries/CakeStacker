<script>

    const canvas = document.getElementById('gameCanvas');

    const ctx = canvas.getContext('2d');

    const startScreen = document.getElementById('startScreen');

    const startBtn = document.getElementById('startBtn');

    const ui = document.getElementById('ui');

    const scoreDisplay = document.getElementById('score');

    const highScoreDisplay = document.getElementById('highScore');

    const container = document.getElementById('gameContainer');

    const perfectPopup = document.getElementById('perfectPopup');


    let gameActive = false;

    let score = 0;

    let highScore = localStorage.getItem('cakeHighScore') ? parseInt(localStorage.getItem('cakeHighScore')) : 0;

    let currentWobble = 0;

    let wobbleSpeed = 0.07;

    

    let claw = {

        x: 200,

        y: 40,

        width: 60,

        speed: 4.0,

        direction: 1

    };


    let activeLayer = null;

    const plate = { x: 130, y: 530, width: 140, height: 15 };

    let stack = [];

    let fireworks = [];


    const cakeColors = ['#ff6b6b', '#feca57', '#ff9ff3', '#1dd1a1', '#fa8231', '#a855f7'];

    const frostingColors = ['#ffffff', '#ffb6c1', '#f5f6fa', '#ffeaa7'];


    highScoreDisplay.innerText = highScore;


    function createNewLayer() {

        if (score >= 30) {

            activeLayer = null; 

            return;

        }

        const width = Math.max(65, 130 - score * 2.5);

        activeLayer = {

            x: claw.x - width / 2,

            y: claw.y + 20,

            width: width,

            height: 35,

            isDropping: false,

            speedY: 12, 

            color: cakeColors[Math.floor(Math.random() * cakeColors.length)],

            frostingColor: frostingColors[Math.floor(Math.random() * frostingColors.length)],

            offset: 0

        };

    }


    function triggerDrop(e) {

        if (!gameActive || !activeLayer || activeLayer.isDropping) return;

        if (e) { e.preventDefault(); e.stopPropagation(); }

        activeLayer.isDropping = true;

    }


    container.addEventListener('touchstart', triggerDrop, { passive: false });

    container.addEventListener('mousedown', triggerDrop);


    function startGame(e) {

        if (e) { e.preventDefault(); e.stopPropagation(); }

        startScreen.style.display = 'none';

        ui.style.display = 'block';

        resetGame();

        gameActive = true;

        animate();

    }


    startBtn.addEventListener('touchend', startGame, { passive: false });

    startBtn.addEventListener('click', startGame);


    function resetGame() {

        score = 0;

        currentWobble = 0;

        scoreDisplay.innerText = score;

        highScoreDisplay.innerText = highScore;

        stack = [];

        fireworks = [];

        claw.speed = 4.0;

        createNewLayer();

    }


    function spawnFirework() {

        const x = Math.random() * canvas.width;

        const y = Math.random() * (canvas.height / 2);

        const colors = ['#ff0055', '#00ffcc', '#ffcc00', '#ff00ff', '#00ff00'];

        const color = colors[Math.floor(Math.random() * colors.length)];

        

        for (let i = 0; i < 30; i++) {

            const angle = Math.random() * Math.PI * 2;

            const velocity = Math.random() * 5 + 2;

            fireworks.push({

                x: x, y: y,

                vx: Math.cos(angle) * velocity,

                vy: Math.sin(angle) * velocity,

                alpha: 1,

                color: color

            });

        }

    }


    function handleFireworks() {

        for (let i = fireworks.length - 1; i >= 0; i--) {

            let p = fireworks[i];

            p.x += p.vx;

            p.y += p.vy;

            p.vy += 0.05; 

            p.alpha -= 0.015;

            

            ctx.save();

            ctx.globalAlpha = p.alpha;

            ctx.fillStyle = p.color;

            ctx.beginPath();

            ctx.arc(p.x, p.y, 3, 0, Math.PI * 2);

            ctx.fill();

            ctx.restore();


            if (p.alpha <= 0) fireworks.splice(i, 1);

        }

        if (Math.random() < 0.05) spawnFirework();

    }


    function updateHighScore() {

        if (score > highScore) {

            highScore = score;

            localStorage.setItem('cakeHighScore', highScore);

            highScoreDisplay.innerText = highScore;

        }

    }


    function animate() {

        ctx.clearRect(0, 0, canvas.width, canvas.height);


        let totalOffset = 0;

        stack.forEach((layer) => {

            totalOffset += layer.offset;

        });

        

        if (stack.length > 0 && score < 30 && Math.abs(totalOffset) > 20) {

            let severity = (Math.abs(totalOffset) - 20) / 25; 

            currentWobble = Math.sin(Date.now() * wobbleSpeed) * (severity * 22);

            

            if (Math.abs(totalOffset) > 45) {

                gameOver(`💥 OH NO! The tower got too un-centered, wobbled out of control, and TOPPLED!`);

                return;

            }

        } else {

            currentWobble = 0; 

        }


        // 1. Draw Plate

        ctx.fillStyle = '#ced6e0';

        ctx.beginPath();

        ctx.ellipse(plate.x + plate.width/2, plate.y + plate.height/2, plate.width/2, plate.height, 0, 0, Math.PI * 2);

        ctx.fill();


        // 2. Move & Draw Claw

        if (activeLayer && !activeLayer.isDropping) {

            claw.x += claw.speed * claw.direction;

            if (claw.x > canvas.width - 40 || claw.x < 40) claw.direction *= -1;

        }

        

        ctx.strokeStyle = '#2f3542';

        ctx.lineWidth = 3;

        ctx.beginPath();

        ctx.moveTo(claw.x, 0);

        ctx.lineTo(claw.x, claw.y);

        ctx.stroke();


        ctx.fillStyle = '#747d8c';

        ctx.fillRect(claw.x - 20, claw.y, 40, 10);

        ctx.fillRect(claw.x - 25, claw.y, 8, 20);

        ctx.fillRect(claw.x + 17, claw.y, 8, 20);


        // 3. Draw Stacked Layers

        stack.forEach((layer, idx) => {

            let wobbleShift = (idx / stack.length) * currentWobble;

            drawCake(layer.x + wobbleShift, layer.y, layer.width, layer.height, layer.color, layer.frostingColor, idx === 29);

        });


        // 4. Handle Fireworks

        if (score >= 30) {

            handleFireworks();

            if (fireworks.length === 0 && gameActive) {

                gameActive = false;

                updateHighScore();

                alert(`🎉 AMAZING! You reached 30 layers and placed the final birthday candle! Happy 8th Birthday! 🎂`);

                startScreen.style.display = 'flex';

                ui.style.display = 'none';

                return;

            }

        }


        // 5. Handle Active Layer Drop Physics

        if (activeLayer) {

            if (!activeLayer.isDropping) {

                activeLayer.x = claw.x - activeLayer.width / 2;

            } else {

                activeLayer.y += activeLayer.speedY;


                let targetY = plate.y - activeLayer.height;

                let targetWidth = plate.width;

                let targetX = plate.x;

                let targetCenterX = plate.x + plate.width / 2;


                if (stack.length > 0) {

                    const topLayer = stack[stack.length - 1];

                    targetY = topLayer.y - activeLayer.height;

                    targetWidth = topLayer.width;

                    targetX = topLayer.x;

                    targetCenterX = topLayer.x + topLayer.width / 2;

                }


                if (activeLayer.y >= targetY) {

                    activeLayer.y = targetY;

                    

                    const layerCenter = activeLayer.x + activeLayer.width / 2;

                    const halfW = activeLayer.width / 2;


                    if (layerCenter + halfW > targetX && layerCenter - halfW < targetX + targetWidth) {

                        let missOffset = layerCenter - targetCenterX;

                        activeLayer.offset = missOffset;


                        if (Math.abs(missOffset) <= 11) {

                            score += 3; 

                            showPerfectIndicator();

                        } else {

                            score += 1; 

                        }


                        stack.push(activeLayer);

                        scoreDisplay.innerText = score;

                        updateHighScore();

                        claw.speed += 0.35; 

                        

                        if (stack.length > 5) {

                            stack.forEach(l => l.y += 35);

                        }

                        

                        if (score >= 30) {

                            spawnFirework();

                        } else {

                            createNewLayer();

                        }

                    } else {

                        gameOver(`💥 BOOM! You completely missed the stack and dropped the cake!`);

                    }

                }

            }

            drawCake(activeLayer.x, activeLayer.y, activeLayer.width, activeLayer.height, activeLayer.color, activeLayer.frostingColor, false);

        }


        if (gameActive || fireworks.length > 0) {

            requestAnimationFrame(animate);

        }

    }


    function showPerfectIndicator() {

        perfectPopup.classList.add('pop');

        setTimeout(() => {

            perfectPopup.classList.remove('pop');

        }, 600);

    }


    function drawCake(x, y, w, h, color, frosting, hasCandle) {

        ctx.fillStyle = color;

        ctx.beginPath();

        ctx.roundRect(x, y, w, h, 6);

        ctx.fill();


        ctx.fillStyle = frosting;

        ctx.fillRect(x, y + 4, w, 8);

        for(let i = 0; i < w; i += 15) {

            ctx.beginPath();

            ctx.arc(x + i + 5, y + 12, 5, 0, Math.PI);

            ctx.fill();

        }


        if (hasCandle) {

            let cx = x + w / 2;

            ctx.fillStyle = '#ff9800';

            ctx.fillRect(cx - 4, y - 20, 8, 20);

            ctx.fillStyle = '#ff5722';

            ctx.beginPath();

            ctx.arc(cx, y - 25, 5, 0, Math.PI * 2);

            ctx.fill();

            ctx.fillStyle = '#ffeb3b';

            ctx.beginPath();

            ctx.arc(cx, y - 23, 2, 0, Math.PI * 2);

            ctx.fill();

        }

    }


    function gameOver(message) {

        gameActive = false;

        updateHighScore();

        alert(`${message}\n🎉 Final Score: ${score} points!`);

        startScreen.style.display = 'flex';

        ui.style.display = 'none';

    }

</script>
